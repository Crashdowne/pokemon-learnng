# SECTION 5: HELPER FUNCTIONS FOR EFFECTIVENESS CALCULATION
# ═══════════════════════════════════════════════════════════════════════════

def get_effectiveness(atk_type: str, def_type: str) -> float:
    """Get effectiveness multiplier for a single attacking type vs single defending type."""
    return TYPE_CHART.get(atk_type, {}).get(def_type, 1.0)

def get_dual_effectiveness(atk_type: str, def_type1: str, def_type2: str) -> float:
    """Get effectiveness multiplier for attacking type vs dual-type defender.
    Formula: multiply effectiveness vs type1 by effectiveness vs type2."""
    return get_effectiveness(atk_type, def_type1) * get_effectiveness(atk_type, def_type2)

def get_defensive_profile_dual(type1: str, type2: str) -> dict:
    """Get the full defensive profile for a dual-type Pokemon.
    Returns dict with keys: 0, 0.25, 0.5, 1, 2, 4 mapping to lists of attacking types."""
    profile = {0: [], 0.25: [], 0.5: [], 1: [], 2: [], 4: []}
    for atk in TYPES:
        eff = get_dual_effectiveness(atk, type1, type2)
        profile[eff].append(atk)
    return profile
